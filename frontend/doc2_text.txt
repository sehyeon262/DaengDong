Part 1. 코틀린 기초 문법 정리
코드를 작성하기 전에, 우리 프로젝트에서 자주 쓰는 코틀린 문법을 정리합니다.
1-1. 변수 선언: val vs var
  val name = &quot;홍길동&quot;      // 변경 불가 (final). 한 번 정하면 끝
  var age = 25             // 변경 가능. 나중에 바꿀 수 있음
  name = &quot;김철수&quot;           // ❌ 에러! val은 변경 불가
  age = 26                 // ✅ OK! var는 변경 가능
💡 우리 프로젝트 규칙
기본적으로 val을 사용하고, 값이 변해야 하는 경우에만 var을 사용합니다.
1-2. 함수 선언: fun
  // 기본 함수
  fun greet(name: String): String {
      return &quot;Hello, $name&quot;
  }
  // 반환값이 없는 함수 (Unit = Java의 void)
  fun printMessage(msg: String) {
      println(msg)
  }
  // 한 줄 함수 (간단할 때)
  fun add(a: Int, b: Int): Int = a + b
1-3. 클래스 선언: class
  // 일반 클래스
  class UserRepository(
      private val api: UserApi       // 생성자에서 바로 프로퍼티 선언
  ) {
      fun getUsers() { /* ... */ }
  }
  // private val = 이 클래스 내부에서만 사용 가능한 변경불가 프로퍼티
💡 private val의 의미
private = 외부에서 접근 불가, val = 변경 불가. 즉, 이 클래스 안에서만 쓰고 바꾸지 않는 값입니다.
1-4. data class (데이터 담는 클래스)
data class는 데이터를 담기만 하는 특별한 클래스입니다. 자동으로 toString, equals, copy 등을 만들어줍니다.
  data class User(
      val id: Int,
      val name: String,
      val email: String
  )
  // 사용
  val user = User(1, &quot;홍길동&quot;, &quot;hong@test.com&quot;)
  println(user.name)    // 홍길동
1-5. suspend 함수 (비동기 처리)
서버 통신은 시간이 걸립니다. suspend는 &quot;이 함수는 시간이 걸릴 수 있으니 기다려라&quot;라는 표시입니다.
  // suspend = &quot;이 함수는 시간이 걸릴 수 있음&quot;
  suspend fun login(email: String, password: String) {
      val response = api.login(...)  // 서버 응답 기다림
      tokenStore.save(response.token) // 저장
  }
⚠️ 규칙
API 호출, DB 접근 등 시간이 걸리는 함수에는 반드시 suspend를 붙입니다. 우리 프로젝트에서는 Repository, UseCase, API의 함수에 모두 suspend를 붙입니다.
1-6. object (싱글톤)
프로그램 전체에서 딱 하나만 존재하는 객체입니다. 상수 모음이나 유틸리티에 사용합니다.
  object Constants {
      const val BASE_URL = &quot;http://10.0.2.2:8080/&quot;
      const val TOKEN_KEY = &quot;access_token&quot;
  }
  object Routes {
      const val LOGIN = &quot;login&quot;
      const val HOME = &quot;home&quot;
  }
  // 사용
  val url = Constants.BASE_URL

1-8. 어노테이션 (@)
코드에 추가 정보를 붙이는 태그입니다. &quot;이 코드는 이런 용도야&quot;라고 프레임워크에게 알려줍니다.
어노테이션
무슨 뜻?
@HiltViewModel
&quot;Hilt아, 이건 ViewModel이야. 네가 관리해줘&quot;
@Inject constructor
&quot;Hilt아, 이 생성자의 파라미터를 네가 자동으로 넣어줘&quot;
@Module
&quot;Hilt아, 여기에 객체 만드는 방법이 적혀있어&quot;
@Provides
&quot;Hilt아, 이 함수가 반환하는 객체를 필요한 곳에 넣어줘&quot;
@Singleton
&quot;앱 전체에서 이 객체는 하나만 만들어&quot;
@Composable
&quot;이 함수는 Compose UI를 그리는 함수야&quot;
@POST / @GET
&quot;Retrofit아, 이 함수는 POST/GET 요청을 보내&quot;
@Body
&quot;Retrofit아, 이 파라미터를 JSON 본문으로 보내&quot;
@AndroidEntryPoint
&quot;Hilt아, 이 Activity에서도 의존성 주입 사용할거야&quot;
@HiltAndroidApp
&quot;Hilt아, 앱 전체에서 너를 초기화할게&quot;
1-9. 널 안전성 (Nullable)
코틀린은 null을 엄격하게 관리합니다. 변수가 null이 될 수 있으면 ?를 붙여야 합니다.
  val name: String = &quot;홍길동&quot;    // null 불가 (항상 값이 있음)
  val error: String? = null      // null 가능 (?가 붙음)
  // null 체크 방법
  error?.let { println(it) }     // error가 null이 아닐 때만 실행
  val length = error?.length ?: 0 // null이면 0 사용
💡 우리 프로젝트에서
State의 error 필드처럼 &quot;있을 수도 없을 수도 있는&quot; 값에 String?을 사용합니다.

⚠️ 필드 이름 규칙

Part 3. interface — API 인터페이스 작성하기
서버와 통신할 방법을 정의하는 파일입니다.
위치: data/remote/ 폴더
작성 방법
Step 1. interface 키워드로 시작
class가 아니라 interface입니다. Retrofit이 자동으로 구현체를 만들어줍니다.
Step 2. 각 함수에 HTTP 메서드 어노테이션 붙이기
  // data/remote/AuthApi.kt
  interface AuthApi {
      // 로그인 요청 (POST = 데이터 보내기)
      @POST(&quot;auth/login&quot;)
      suspend fun login(
          @Body request: LoginRequest  // JSON 본문으로 보냄
      ): LoginResponse                 // 서버 응답을 이 타입으로 받음
      // 회원가입 (POST = 데이터 보내기)
      @POST(&quot;auth/register&quot;)
      suspend fun register(
          @Body request: RegisterRequest
      ): RegisterResponse
      // 사용자 목록 조회 (GET = 데이터 가져오기)
      @GET(&quot;users&quot;)
      suspend fun getUsers(): List&lt;User&gt;
      // 특정 사용자 조회 (URL에 id 넣기)
      @GET(&quot;users/{id}&quot;)
      suspend fun getUser(
          @Path(&quot;id&quot;) userId: Int    // URL의 {id} 자리에 값 넣음
      ): User
  }
💡 함수가 뭔 하는지 한 줄 요약
@POST(&quot;auth/login&quot;) + @Body request: LoginRequest → &quot;auth/login 주소로 LoginRequest를 JSON으로 보내고, 응답을 LoginResponse로 받아라&quot;
Part 4. Retrofit 어노테이션 상세 설명
각 어노테이션이 실제로 뭔 하는지 예제로 보여드립니다.
@POST — 서버에 데이터 보내기
  @POST(&quot;auth/login&quot;)          // → POST http://서버주소/auth/login
  suspend fun login(
      @Body request: LoginRequest
  ): LoginResponse
실제로 보내지는 HTTP 요청:
  POST /auth/login
  Content-Type: application/json
  {
    &quot;email&quot;: &quot;test@test.com&quot;,
    &quot;password&quot;: &quot;1234&quot;
  }
@GET — 서버에서 데이터 가져오기
  @GET(&quot;users&quot;)               // → GET http://서버주소/users
  suspend fun getUsers(): List&lt;User&gt;
@Path — URL에 값 넣기
  @GET(&quot;users/{id}&quot;)           // {id} 자리에 값이 들어감
  suspend fun getUser(
      @Path(&quot;id&quot;) userId: Int   // userId = 3 → GET /users/3
  ): User
@Query — URL 뒤에 ?파라미터 붙이기
  @GET(&quot;posts&quot;)               // → GET /posts?page=1&amp;size=20
  suspend fun getPosts(
      @Query(&quot;page&quot;) page: Int,
      @Query(&quot;size&quot;) size: Int
  ): List&lt;Post&gt;
@PUT — 데이터 수정
  @PUT(&quot;users/{id}&quot;)
  suspend fun updateUser(
      @Path(&quot;id&quot;) userId: Int,
      @Body request: UpdateUserRequest
  ): User
@DELETE — 데이터 삭제
  @DELETE(&quot;posts/{id}&quot;)
  suspend fun deletePost(
      @Path(&quot;id&quot;) postId: Int
  )
@Header — 인증 토큰 보내기
  @GET(&quot;profile&quot;)
  suspend fun getProfile(
      @Header(&quot;Authorization&quot;) token: String
      // → Header에 &quot;Authorization: Bearer eyJ...&quot; 추가
  ): User

@Inject constructor
 = Hilt이 AuthApi와 TokenDataStore를 자동으로 넣어줍니다.
Step 2. 기능 함수 작성

Part 6. UseCase — 비즈니스 로직 작성하기
위치: domain/usecase/ 폴더
UseCase는 &quot;기능 하나&quot;를 담당합니다. Repository를 호출하는 것이 전부입니다.
작성 방법
Step 1. 클래스 선언 + Repository 받기
  class LoginUseCase @Inject constructor(
      private val repository: AuthRepository
  ) {
Step 2. operator fun invoke 작성
invoke를 쓰면 함수처럼 호출할 수 있습니다.
      suspend operator fun invoke(email: String, password: String) {

      }

전체 코드

  class LoginUseCase @Inject constructor(
      private val repository: AuthRepository
  ) {
      suspend operator fun invoke(
          email: String,
          password: String
      ) {
          repository.login(email, password)
      }
  }
invoke의 효과
❌ invoke 없이
loginUseCase.execute(email, pw)
✅ invoke 사용
loginUseCase(email, pw)
💡 항상 기억
UseCase 하나 = 기능 하나. LoginUseCase는 로그인만, GetUsersUseCase는 사용자 목록 가져오기만 담당합니다.

Part 8. ViewModel — 상태 관리자 작성하기
위치: ui/screen/[feature]/ 폴더
ViewModel은 화면의 &quot;리모콘&quot;입니다. UseCase를 호출하고 결과에 따라 State를 업데이트합니다.
작성 방법 (한 줄씩 설명)
Step 1. 클래스 선언
  @HiltViewModel                              // Hilt에게 &quot;이건 ViewModel이야&quot; 알려줌
  class LoginViewModel @Inject constructor(   // Hilt가 UseCase를 자동 주입
      private val loginUseCase: LoginUseCase
  ) : ViewModel() {                           // Android ViewModel 상속
Step 2. State 선언
      // 내부용 (ViewModel 안에서만 변경 가능)
      private val _state = MutableStateFlow(LoginState())
      // 외부용 (Screen에서 읽기만 가능)
      val state = _state.asStateFlow()
왜 두 개인가요?
_state는 ViewModel 안에서만 바꿀 수 있고, state는 Screen에서 읽기만 합니다. 이렇게 하면 Screen이 직접 상태를 바꾸지 못합니다.
Step 3. 기능 함수 작성
      fun login(email: String, password: String) {
          // viewModelScope.launch = &quot;비동기로 실행해라&quot;
          viewModelScope.launch {
              // 1단계: 로딩 시작
              _state.update { it.copy(isLoading = true) }
              try {
                  // 2단계: UseCase 호출 (서버 통신)
                  loginUseCase(email, password)
                  // 3단계: 성공
                  _state.update { it.copy(isLoading = false, isSuccess = true) }
              } catch (e: Exception) {
                  // 4단계: 실패
                  _state.update { it.copy(isLoading = false, error = e.message) }
              }
          }
      }
  }
it.copy() 설명
data class의 copy()는 &quot;기존 값은 유지하고, 지정한 것만 변경&quot;합니다.
  // 현재 상태: LoginState(isLoading=false, isSuccess=false, error=null)
  it.copy(isLoading = true)
  // 결과: LoginState(isLoading=true, isSuccess=false, error=null)
  // isLoading만 true로 변경됨, 나머지는 그대로
viewModelScope.launch 설명
launch는 &quot;새 작업을 백그라운드에서 실행해라&quot;입니다. 서버 통신 중에도 앱이 멈추지 않습니다.
viewModelScope = ViewModel이 살아있는 동안만 작업을 실행합니다. 화면이 닫히면 자동 취소됩니다.

          )
          Spacer(modifier = Modifier.height(16.dp))

Part 12. DI (Hilt) — 의존성 주입 설정
위치: di/ 폴더
Hilt가 하는 일 (비유)
LoginViewModel을 만들려면 LoginUseCase가 필요하고, LoginUseCase를 만들려면 AuthRepository가 필요하고...
Hilt가 없으면 우리가 직접 다 만들어야 합니다:
  // Hilt 없으면 이렇게 해야 함 (무시무시하게 길어짐)
  val retrofit = Retrofit.Builder()...
  val api = retrofit.create(AuthApi::class.java)
  val dataStore = TokenDataStore(...)
  val repository = AuthRepository(api, dataStore)
  val useCase = LoginUseCase(repository)
  val viewModel = LoginViewModel(useCase)  // 드디어 완성!
Hilt를 쓰면 이 과정이 전부 자동입니다.
Module 작성법
NetworkModule.kt — Retrofit + API 설정
  // di/NetworkModule.kt
  @Module
  @InstallIn(SingletonComponent::class)  // 앱 전체에서 1개만 존재
  object NetworkModule {
      @Provides
      @Singleton
      fun provideRetrofit(): Retrofit {
          return Retrofit.Builder()
              .baseUrl(Constants.BASE_URL)            // 서버 주소
              .addConverterFactory(                    // JSON 변환기
                  GsonConverterFactory.create()
              )
              .build()
      }
      @Provides
      @Singleton
      fun provideAuthApi(retrofit: Retrofit): AuthApi {
          return retrofit.create(AuthApi::class.java)
          // Retrofit이 AuthApi interface의 구현체를 자동 생성
      }
  }
RepositoryModule.kt
  // di/RepositoryModule.kt
  @Module
  @InstallIn(SingletonComponent::class)
  object RepositoryModule {
      @Provides
      @Singleton
      fun provideAuthRepository(
          api: AuthApi,                    // Hilt가 NetworkModule에서 가져옴
          tokenDataStore: TokenDataStore    // Hilt가 자동 생성
      ): AuthRepository {
          return AuthRepository(api, tokenDataStore)
      }
  }
💡 Hilt 흐름 정리
Module에서 @Provides로 객체 생성법을 정의 → @Inject constructor가 있는 클래스에 자동 주입 → @HiltViewModel의 생성자에도 자동 주입

      private val tokenDataStore: TokenDataStore

          val response = api.login(LoginRequest(email, password))

⑥ ViewModel 작성
  // ui/screen/register/RegisterViewModel.kt
  @HiltViewModel
  class RegisterViewModel @Inject constructor(
      private val registerUseCase: RegisterUseCase
  ) : ViewModel() {
      private val _state = MutableStateFlow(RegisterState())
      val state = _state.asStateFlow()
      fun register(name: String, email: String, password: String) {
          viewModelScope.launch {
              _state.update { it.copy(isLoading = true) }
              try {
                  registerUseCase(name, email, password)
                  _state.update { it.copy(isLoading = false, isSuccess = true) }
              } catch (e: Exception) {
                  _state.update { it.copy(isLoading = false, error = e.message) }
              }
          }
      }
  }
